package com.gym.management.service.impl;

import com.gym.management.constant.PaymentMethodType;
import com.gym.management.constant.PaymentStatus;
import com.gym.management.dto.response.PaymentTransactionResponse;
import com.gym.management.entity.PaymentTransaction;
import com.gym.management.entity.User;

import com.gym.management.exception.ErrorCode;

import com.gym.management.service.PaymentService;
import com.gym.management.service.event.PaymentConfirmedEvent;
import lombok.RequiredArgsConstructor;
import org.springframework.context.ApplicationEventPublisher;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * Agent 2 - Membership, Package & Payment Module.
 *
 * Lưu ý về phụ thuộc: PaymentServiceImpl KHÔNG phụ thuộc MembershipService. Trước đây 2 bên inject
 * constructor lẫn nhau (Membership → Payment để tạo giao dịch PENDING lúc đăng ký/nâng/hạ cấp;
 * Payment → Membership để kích hoạt gói trong confirmPayment) nên ứng dụng không khởi động được —
 * Spring Boot cấm vòng phụ thuộc bean theo mặc định (nhận định cũ ở đây rằng "Spring xử lý được qua
 * proxy" là SAI, mvn verify với Mockito không bắt được lỗi này).
 *
 * Nay chỉ còn 1 chiều Membership → Payment. Khi xác nhận xong 1 giao dịch, Payment phát
 * {@link PaymentConfirmedEvent} (đồng bộ, chung transaction) và
 * {@link com.gym.management.service.listener.MembershipPaymentListener} kích hoạt gói. Module khác
 * (vd POS của Agent 5) lắng nghe cùng sự kiện này và lọc theo relatedType.
 * KHÔNG thêm lại phụ thuộc ngược Payment → Membership và KHÔNG bật spring.main.allow-circular-references.
 */
@Service
@RequiredArgsConstructor
public class PaymentServiceImpl implements PaymentService {

    private final PaymentTransactionRepository paymentTransactionRepository;
    private final ApplicationEventPublisher eventPublisher;

    @Override
    @Transactional
    public PaymentTransactionResponse createPendingTransaction(User user, String relatedType, UUID relatedId, BigDecimal amount) {
        PaymentTransaction tx = new PaymentTransaction();
        tx.setUser(user);
        tx.setRelatedType(relatedType);
        tx.setRelatedId(relatedId);
        tx.setAmount(amount);
        tx.setPaymentMethod(PaymentMethodType.CASH); // mặc định, có thể đổi lúc confirm
        tx.setStatus(PaymentStatus.PENDING);
        tx.setTransactionCode(generateTransactionCode());
        return toResponse(saveWithRetryOnCodeCollision(tx));
    }

    @Override
    @Transactional
    public PaymentTransactionResponse confirmPayment(UUID transactionId, String paymentMethod) {
        PaymentTransaction tx = paymentTransactionRepository.findById(transactionId)
                .orElseThrow(() -> new BusinessException(ErrorCode.PAYMENT_NOT_FOUND));

        if (PaymentStatus.SUCCESS.equals(tx.getStatus())) {
            return toResponse(tx); // idempotent — xác nhận lần 2 không lỗi, không phát lại sự kiện, không lặp side-effect
        }

        if (paymentMethod != null && !paymentMethod.isBlank()) {
            tx.setPaymentMethod(paymentMethod);
        }
        tx.setStatus(PaymentStatus.SUCCESS);
        tx = paymentTransactionRepository.save(tx);

        // Điểm nối DUY NHẤT giữa Payment và các module khác — Payment không gọi thẳng Membership (tránh vòng
        // phụ thuộc bean), chỉ phát sự kiện đồng bộ chung transaction:
        //  - MEMBERSHIP: MembershipPaymentListener kích hoạt gói (PENDING → ACTIVE).
        //  - POS: Agent 5 có thể lắng nghe cùng sự kiện để trừ kho ở module riêng; module này chỉ đảm bảo
        //    giao dịch được ghi log và đổi trạng thái đúng.
        eventPublisher.publishEvent(new PaymentConfirmedEvent(tx.getId(), tx.getRelatedType(), tx.getRelatedId()));

        return toResponse(tx);
    }

    @Override
    public PaymentTransaction getEntityById(UUID transactionId) {
        return paymentTransactionRepository.findById(transactionId)
                .orElseThrow(() -> new BusinessException(ErrorCode.PAYMENT_NOT_FOUND));
    }

    @Override
    public List<PaymentTransactionResponse> getMyPayments(UUID userId) {
        return paymentTransactionRepository.findByUserIdOrderByCreatedAtDesc(userId).stream()
                .map(this::toResponse)
                .collect(Collectors.toList());
    }

    @Override
    public com.gym.management.dto.response.PageResponse<PaymentTransactionResponse> getMyPayments(
            UUID userId, org.springframework.data.domain.Pageable pageable) {
        return com.gym.management.dto.response.PageResponse.from(
                paymentTransactionRepository.findByUserId(userId, pageable), this::toResponse);
    }

    @Override
    public com.gym.management.dto.response.PageResponse<PaymentTransactionResponse> getAllPayments(
            java.time.LocalDate from, java.time.LocalDate to, org.springframework.data.domain.Pageable pageable) {
        java.time.LocalDate effectiveFrom = from != null ? from : java.time.LocalDate.now();
        java.time.LocalDate effectiveTo = to != null ? to : java.time.LocalDate.now();
        java.time.LocalDateTime fromDateTime = effectiveFrom.atStartOfDay();
        java.time.LocalDateTime toDateTime = effectiveTo.plusDays(1).atStartOfDay(); // exclusive upper bound, gồm trọn ngày "to"
        return com.gym.management.dto.response.PageResponse.from(
                paymentTransactionRepository.findByCreatedAtBetween(fromDateTime, toDateTime, pageable), this::toResponse);
    }

    private PaymentTransaction saveWithRetryOnCodeCollision(PaymentTransaction tx) {
        try {
            return paymentTransactionRepository.save(tx);
        } catch (DataIntegrityViolationException ex) {
            // transaction_code trùng (xác suất cực thấp với UUID) — thử lại 1 lần với code mới.
            tx.setTransactionCode(generateTransactionCode());
            try {
                return paymentTransactionRepository.save(tx);
            } catch (DataIntegrityViolationException ex2) {
                throw new BusinessException(ErrorCode.PAYMENT_DUPLICATE_TRANSACTION);
            }
        }
    }

    private String generateTransactionCode() {
        return "TX-" + System.currentTimeMillis() + "-" + UUID.randomUUID().toString().substring(0, 8).toUpperCase();
    }

    private PaymentTransactionResponse toResponse(PaymentTransaction tx) {
        return PaymentTransactionResponse.builder()
                .id(tx.getId())
                .userId(tx.getUser().getId())
                .relatedType(tx.getRelatedType())
                .relatedId(tx.getRelatedId())
                .amount(tx.getAmount())
                .paymentMethod(tx.getPaymentMethod())
                .status(tx.getStatus())
                .transactionCode(tx.getTransactionCode())
                .createdAt(tx.getCreatedAt())
                .build();
    }
}

/**
 * Test cho PaymentService — đặc biệt {@link com.gym.management.service.PaymentService#confirmPayment}
 * vì đây là điểm nối DUY NHẤT giữa Payment và các module khác: Payment KHÔNG gọi thẳng MembershipService
 * (tránh vòng phụ thuộc bean, xem Javadoc PaymentServiceImpl) mà phát {@link PaymentConfirmedEvent}.
 * Việc listener kích hoạt đúng gói được test ở MembershipPaymentListenerTest và MembershipPaymentWiringTest.
 */
@ExtendWith(MockitoExtension.class)
class PaymentServiceImplTest {

    @Mock
    private PaymentTransactionRepository paymentTransactionRepository;
    @Mock
    private ApplicationEventPublisher eventPublisher;

    private PaymentServiceImpl paymentService;

    @BeforeEach
    void setUp() {
        paymentService = new PaymentServiceImpl(paymentTransactionRepository, eventPublisher);
    }

    private PaymentTransaction pendingMembershipTransaction(UUID memberPackageId) {
        PaymentTransaction tx = new PaymentTransaction();
        tx.setId(UUID.randomUUID());
        User user = new User();
        user.setId(UUID.randomUUID());
        tx.setUser(user);
        tx.setRelatedType(RelatedType.MEMBERSHIP);
        tx.setRelatedId(memberPackageId);
        tx.setAmount(new BigDecimal("500000"));
        tx.setPaymentMethod("CASH");
        tx.setStatus(PaymentStatus.PENDING);
        tx.setTransactionCode("TX-TEST-0001");
        return tx;
    }

    @Test
    void confirmPayment_shouldPublishEventAfterSaving_whenRelatedTypeIsMembership() {
        UUID memberPackageId = UUID.randomUUID();
        PaymentTransaction tx = pendingMembershipTransaction(memberPackageId);

        when(paymentTransactionRepository.findById(tx.getId())).thenReturn(Optional.of(tx));
        when(paymentTransactionRepository.save(any(PaymentTransaction.class)))
                .thenAnswer(invocation -> invocation.getArgument(0));

        var response = paymentService.confirmPayment(tx.getId(), "CASH");

        assertThat(response.getStatus()).isEqualTo(PaymentStatus.SUCCESS);
        // Điểm nối bắt buộc: confirm 1 giao dịch MEMBERSHIP PHẢI phát sự kiện mang đúng member_package cần kích hoạt,
        // và phải phát SAU khi đã lưu trạng thái SUCCESS (listener chạy đồng bộ, chung transaction).
        InOrder order = inOrder(paymentTransactionRepository, eventPublisher);
        order.verify(paymentTransactionRepository).save(any(PaymentTransaction.class));
        order.verify(eventPublisher).publishEvent(
                new PaymentConfirmedEvent(tx.getId(), RelatedType.MEMBERSHIP, memberPackageId));
    }

    @Test
    void confirmPayment_shouldBeIdempotent_whenAlreadySuccess() {
        UUID memberPackageId = UUID.randomUUID();
        PaymentTransaction tx = pendingMembershipTransaction(memberPackageId);
        tx.setStatus(PaymentStatus.SUCCESS); // đã confirm từ trước

        when(paymentTransactionRepository.findById(tx.getId())).thenReturn(Optional.of(tx));

        var response = paymentService.confirmPayment(tx.getId(), "CASH");

        assertThat(response.getStatus()).isEqualTo(PaymentStatus.SUCCESS);
        // Gọi lại lần 2 KHÔNG được phát sự kiện lần nữa (nếu không gói sẽ bị kích hoạt lặp / side-effect lặp)
        verify(eventPublisher, never()).publishEvent(any(PaymentConfirmedEvent.class));
        verify(paymentTransactionRepository, never()).save(any());
    }

    @Test
    void confirmPayment_shouldPublishEventWithPosType_soOtherModulesCanReact() {
        PaymentTransaction tx = pendingMembershipTransaction(UUID.randomUUID());
        UUID posOrderId = UUID.randomUUID();
        tx.setRelatedType(RelatedType.POS); // giao dịch POS (Agent 5), không phải Membership
        tx.setRelatedId(posOrderId);

        when(paymentTransactionRepository.findById(tx.getId())).thenReturn(Optional.of(tx));
        when(paymentTransactionRepository.save(any(PaymentTransaction.class)))
                .thenAnswer(invocation -> invocation.getArgument(0));

        paymentService.confirmPayment(tx.getId(), null);

        // Payment không phân biệt loại giao dịch: luôn phát sự kiện, MembershipPaymentListener tự bỏ qua POS
        // (xem MembershipPaymentListenerTest), còn Agent 5 có thể lắng nghe cùng sự kiện để trừ kho.
        verify(eventPublisher).publishEvent(new PaymentConfirmedEvent(tx.getId(), RelatedType.POS, posOrderId));
    }

    @Test
    void confirmPayment_shouldThrow_whenTransactionNotFound() {
        UUID missingId = UUID.randomUUID();
        when(paymentTransactionRepository.findById(missingId)).thenReturn(Optional.empty());

        assertThatThrownBy(() -> paymentService.confirmPayment(missingId, "CASH"))
                .isInstanceOf(BusinessException.class)
                .extracting("errorCode")
                .isEqualTo(ErrorCode.PAYMENT_NOT_FOUND);
        verifyNoInteractions(eventPublisher);
    }

/**
 * Cung cấp bean `Clock` cố định timezone Asia/Ho_Chi_Minh, dùng thay cho gọi trực tiếp
 * `LocalDateTime.now()` trong `BookingServiceImpl`/`TrainerScheduleServiceImpl` — mục đích
 * DUY NHẤT là để unit test giả lập thời gian bằng `Clock.fixed(...)` cho lazy auto-status
 * (Agent1/Phi yêu cầu ở "Việc còn thiếu" mục 6).
 *
 * ⚠️ Đặt tạm trong package `service/` (không phải `config/`) vì `config/` thuộc sở hữu
 * Agent 1 (ARCHITECTURE.md mục 2). Nếu Agent 1 đã có sẵn 1 bean `Clock` khác ở nơi khác
 * (trùng tên `clock`), Spring sẽ báo lỗi "duplicate bean definition" khi merge — chỉ cần XOÁ
 * FILE NÀY và dùng bean có sẵn, không cần sửa gì ở BookingServiceImpl/TrainerScheduleServiceImpl
 * (cả 2 chỉ inject theo type `Clock`, không quan tâm bean định nghĩa ở đâu).

@Configuration
public class BookingClockConfig {

    @Bean
    public Clock clock() {
        return Clock.system(ZoneId.of("Asia/Ho_Chi_Minh"));
    }

/**
 * Agent 6 - Truy vấn báo cáo, CHỈ ĐỌC. Dùng SQL thuần trên bảng (không phụ thuộc entity
 * của module khác). Ngày/giờ là giá trị lưu trong DB, coi như giờ Asia/Ho_Chi_Minh.
 */
@Repository
@RequiredArgsConstructor
public class ReportRepository {

    private final NamedParameterJdbcTemplate jdbc;

    public record RevenueRow(LocalDate day, String relatedType, BigDecimal total) {
    }

    public record RenewalCounts(long expired, long renewed) {
    }

    public record TrafficRow(LocalDate day, int hour, long count) {
    }

    public record TopPackageRow(UUID packageId, String packageName, long registrations, BigDecimal revenue) {
    }

    /** Doanh thu theo ngày và loại. Ngày ghi nhận = updated_at của giao dịch SUCCESS (fallback created_at). */
    public List<RevenueRow> revenueByDay(LocalDateTime from, LocalDateTime toExclusive) {
        String sql = """
                SELECT CAST(COALESCE(updated_at, created_at) AS DATE) AS day_value,
                       related_type,
                       SUM(amount) AS total
                FROM payment_transactions
                WHERE status = 'SUCCESS'
                  AND COALESCE(updated_at, created_at) >= :from
                  AND COALESCE(updated_at, created_at) < :to
                GROUP BY CAST(COALESCE(updated_at, created_at) AS DATE), related_type
                """;
        MapSqlParameterSource params = new MapSqlParameterSource()
                .addValue("from", Timestamp.valueOf(from))
                .addValue("to", Timestamp.valueOf(toExclusive));
        return jdbc.query(sql, params, (rs, i) -> new RevenueRow(
                rs.getDate("day_value").toLocalDate(),
                rs.getString("related_type"),
                rs.getBigDecimal("total")));
    }

    /**
     * Số gói hết hạn trong kỳ và số gói đã gia hạn (cùng user có gói khác, bắt đầu sau gói cũ
     * và không muộn hơn end_date + windowDays). Tính theo end_date vì status chỉ cập nhật lazy.
     * Bọc bằng bảng dẫn xuất vì SQL Server không cho SUM trực tiếp trên biểu thức chứa subquery.
     */
    public RenewalCounts renewalCounts(LocalDate from, LocalDate to, LocalDate today, int windowDays) {
        String sql = """
                SELECT COUNT(*) AS expired_count, COALESCE(SUM(t.renewed), 0) AS renewed_count
                FROM (
                    SELECT CASE WHEN EXISTS (
                               SELECT 1 FROM member_packages m2
                               WHERE m2.user_id = m1.user_id
                                 AND m2.id <> m1.id
                                 AND m2.status IN ('ACTIVE', 'EXPIRED', 'SCHEDULED')
                                 AND m2.start_date > m1.start_date
                                 AND m2.start_date <= DATEADD(DAY, :window, m1.end_date)
                           ) THEN 1 ELSE 0 END AS renewed
                    FROM member_packages m1
                    WHERE m1.end_date IS NOT NULL
                      AND m1.end_date >= :from
                      AND m1.end_date <= :to
                      AND m1.end_date < :today
                      AND m1.status NOT IN ('PENDING', 'CANCELLED', 'SCHEDULED')
                ) t
                """;
        MapSqlParameterSource params = new MapSqlParameterSource()
                .addValue("window", windowDays)
                .addValue("from", Date.valueOf(from))
                .addValue("to", Date.valueOf(to))
                .addValue("today", Date.valueOf(today));
        return jdbc.queryForObject(sql, params, (rs, i) ->
                new RenewalCounts(rs.getLong("expired_count"), rs.getLong("renewed_count")));
    }

    /** Lượt học viên quét mặt vào buổi PT, theo ngày và giờ. */
    public List<TrafficRow> checkinTraffic(LocalDateTime from, LocalDateTime toExclusive) {
        String sql = """
                SELECT CAST(check_in_time AS DATE) AS day_value,
                       DATEPART(HOUR, check_in_time) AS hour_value,
                       COUNT(*) AS cnt
                FROM bookings
                WHERE check_in_time IS NOT NULL
                  AND check_in_time >= :from
                  AND check_in_time < :to
                GROUP BY CAST(check_in_time AS DATE), DATEPART(HOUR, check_in_time)
                """;
        MapSqlParameterSource params = new MapSqlParameterSource()
                .addValue("from", Timestamp.valueOf(from))
                .addValue("to", Timestamp.valueOf(toExclusive));
        return jdbc.query(sql, params, (rs, i) -> new TrafficRow(
                rs.getDate("day_value").toLocalDate(),
                rs.getInt("hour_value"),
                rs.getLong("cnt")));
    }

    /** Gói bán chạy: mỗi giao dịch MEMBERSHIP SUCCESS tính một lượt đăng ký. */
    public List<TopPackageRow> topPackages(LocalDateTime from, LocalDateTime toExclusive, int limit) {
        String sql = """
                SELECT TOP (:limit)
                       p.id AS package_id,
                       p.name AS package_name,
                       COUNT(*) AS registrations,
                       SUM(pt.amount) AS revenue
                FROM payment_transactions pt
                JOIN member_packages mp ON mp.id = pt.related_id
                JOIN packages p ON p.id = mp.package_id
                WHERE pt.related_type = 'MEMBERSHIP'
                  AND pt.status = 'SUCCESS'
                  AND COALESCE(pt.updated_at, pt.created_at) >= :from
                  AND COALESCE(pt.updated_at, pt.created_at) < :to
                GROUP BY p.id, p.name
                ORDER BY COUNT(*) DESC, SUM(pt.amount) DESC, p.name ASC
                """;
        MapSqlParameterSource params = new MapSqlParameterSource()
                .addValue("limit", limit)
                .addValue("from", Timestamp.valueOf(from))
                .addValue("to", Timestamp.valueOf(toExclusive));
        return jdbc.query(sql, params, (rs, i) -> new TopPackageRow(
                UUID.fromString(rs.getString("package_id")),
                rs.getString("package_name"),
                rs.getLong("registrations"),
                rs.getBigDecimal("revenue")));
    }
}
